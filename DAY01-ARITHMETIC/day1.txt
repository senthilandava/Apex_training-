



def get_number(prompt):
    """Safely get a float number from user input."""
    while True:
        try:
            return float(input(prompt))
        except ValueError:
            print("❌ Invalid input. Please enter a valid number.")

def safe_division(a, b):
    """Performs division and handles division by zero."""
    try:
        return a / b
    except ZeroDivisionError:
        return "Undefined (division by zero)"

def safe_floor_division(a, b):
    """Performs floor division and handles division by zero."""
    try:
        return a // b
    except ZeroDivisionError:
        return "Undefined (division by zero)"

def safe_modulus(a, b):
    """Performs modulus and handles division by zero."""
    try:
        return a % b
    except ZeroDivisionError:
        return "Undefined (division by zero)"

def main():
    print("=== Python Arithmetic Operators Demo ===\n")
    
    # Get user inputs
    num1 = get_number("Enter the first number: ")
    num2 = get_number("Enter the second number: ")
    
    print("\n--- Results ---")
    print(f"{num1} + {num2} = {num1 + num2}")       # Addition
    print(f"{num1} - {num2} = {num1 - num2}")       # Subtraction
    print(f"{num1} * {num2} = {num1 * num2}")       # Multiplication
    print(f"{num1} / {num2} = {safe_division(num1, num2)}")  # Division
    print(f"{num1} // {num2} = {safe_floor_division(num1, num2)}")  # Floor Division
    print(f"{num1} % {num2} = {safe_modulus(num1, num2)}")   # Modulus
    print(f"{num1} ** {num2} = {num1 ** num2}")     # Exponentiation
    
    print("\n=== End of Program ===")

if __name__ == "__main__":
    main()
